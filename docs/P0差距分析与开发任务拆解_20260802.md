### 3.1 API差距逐条分析

#### API#1 GET /api/v1/scene/entries（场景词条列表）

| 维度 | 现状 | 目标 | 差距 | 工作量 |
|------|------|------|------|--------|
| 筛选维度 | domain + subScene | clientType + stage + layer | **需新增3个筛选维度**，且clientType/stage是数组包含匹配 | **M** |
| 分组返回 | 无分组，平铺entries | 按layer分4组（dao/fa/shu/qi），每组含label+entries | **需新增分组逻辑** | **S** |
| 返回字段 | 全量字段 | 轻量字段（id/name/oneLineAnswer/entryType/severity/priority/consumerQ） | **需新增字段选择** | **S** |
| 统计信息 | total+returned | summary(total+byLayer) | **需新增统计** | **S** |
| 路由 | `/api/entries` | `/api/v1/scene/entries` | **新路由**，旧路由保留兼容 | **S** |

**关键实现难点**：clientType是数组字段（`tags.clientType[]`），需用Array.includes()匹配；stage有2134条是数组，需同样处理。

**代码示例（Cloudflare Workers）**：
```javascript
// /api/v1/scene/entries 核心逻辑
function filterByScene(entries, clientType, stage, layer) {
  return entries.filter(e => {
    const ct = e.tags?.clientType;
    const st = e.tags?.stage;
    // clientType匹配：数组包含 or 单值相等
    const ctMatch = Array.isArray(ct) ? ct.includes(clientType) : ct === clientType;
    // stage匹配（同上，且支持数组）
    const stMatch = Array.isArray(st) ? st.includes(stage) : st === stage;
    // layer匹配（可选）
    const ly = e.tags?.layer;
    const lyMatch = !layer || (Array.isArray(ly) ? ly.includes(layer) : ly === layer);
    return ctMatch && stMatch && lyMatch;
  });
}

function groupByLayer(entries) {
  const layerOrder = ['dao', 'fa', 'shu', 'qi'];
  const layerLabels = { dao: '道·心里懂的', fa: '法·当场走的', shu: '术·嘴上说的', qi: '器·递给客户的' };
  return layerOrder.map(layer => ({
    layer,
    label: layerLabels[layer],
    entries: entries.filter(e => {
      const ly = e.tags?.layer;
      return Array.isArray(ly) ? ly.includes(layer) : ly === layer;
    }).sort(sortByPriority).map(pickSlimFields)
  })).filter(g => g.entries.length > 0);
}
```

**代码示例**：
```javascript
// /api/v1/entry/:id 核心逻辑
async function handleEntryDetail(id, allEntries, entryMap) {
  const entry = entryMap.get(id);
  if (!entry) return { code: 1002, message: '词条不存在' };

  // 解析relatedEntries: ID[] → EntryObject[]
  const related = (entry.relatedEntries || []).slice(0, 10)
    .map(rid => entryMap.get(rid))
    .filter(Boolean)
    .map(r => ({
      id: r.id, name: r.name, relation: 'sameSubScene',
      entryType: r.entryType, priority: r.priority,
      tags: { layer: r.tags?.layer }
    }));

  // 计算qi层关联（当前非qi层时）
  const qiEntries = [];
  if (entry.tags?.layer !== 'qi') {
    const subScene = entry.subScene;
    // 查同subScene+layer=qi的词条
    qiEntries.push(...allEntries.filter(e =>
      e.subScene === subScene &&
      (Array.isArray(e.tags?.layer) ? e.tags.layer.includes('qi') : e.tags?.layer === 'qi')
    ).slice(0, 3).map(pickSlimFields));
  }

  return { code: 0, data: { ...formatEntry(entry), relatedEntries: related, qiEntries } };
}
```

#### API#4 GET /api/v1/entry/:id/related（词条关联）

| 维度 | 现状 | 目标 | 差距 | 工作量 |
|------|------|------|------|--------|
| 关联算法 | 无 | 5种：同subScene(10)/同legalRef(7)/同ct+stage+layer(5)/同type+severity(3)/道法术器关联(8) | **全新实现** | **L** |
| 关联排序 | 无 | 按权重降序→priority升序 | **新建** | **S** |
| 层级路径 | 无 | layerPath(current/next/nextEntries) | **新建** | **S** |

**结论**：**全新API**，5种关联算法中最复杂的是legalRef交集计算和道法术器关联路径，工作量**L**。

**代码示例**：
```javascript
// 关联算法核心
function computeRelated(entry, allEntries, entryMap, limit = 10) {
  const candidates = new Map(); // id → { weight, relation, entry }

  // 1. 同subScene（权重10）
  allEntries.forEach(e => {
    if (e.id !== entry.id && e.subScene === entry.subScene) {
      candidates.set(e.id, { weight: 10, relation: 'sameSubScene', entry: e });
    }
  });

  // 2. 同legalRef交集（权重7）
  const myLegalRefs = new Set(Array.isArray(entry.legalRef) ? entry.legalRef : [entry.legalRef]);
  allEntries.forEach(e => {
    if (e.id !== entry.id) {
      const eRefs = new Set(Array.isArray(e.legalRef) ? e.legalRef : [e.legalRef]);
      const overlap = [...myLegalRefs].some(r => eRefs.has(r));
      if (overlap && (!candidates.has(e.id) || candidates.get(e.id).weight < 7)) {
        candidates.set(e.id, { weight: 7, relation: 'sameLegalRef', entry: e });
      }
    }
  });

  // 3. 同clientType+stage+layer（权重5）
  // 4. 同entryType+severity（权重3）
  // 5. 道法术器关联（权重8）
  const layerSeq = ['dao', 'fa', 'shu', 'qi'];
  const currentLayer = entry.tags?.layer;
  const currentIdx = layerSeq.indexOf(currentLayer);
  if (currentIdx >= 0) {
    const nextLayer = currentIdx < 3 ? layerSeq[currentIdx + 1] : layerSeq[currentIdx - 1];
    allEntries.forEach(e => {
      if (e.id !== entry.id && e.subScene === entry.subScene) {
        const eLayer = e.tags?.layer;
        if (eLayer === nextLayer || (Array.isArray(eLayer) && eLayer.includes(nextLayer))) {
          if (!candidates.has(e.id) || candidates.get(e.id).weight < 8) {
            candidates.set(e.id, { weight: 8, relation: 'layerPath', entry: e });
          }
        }
      }
    });
  }

  // 排序：权重降序→priority升序→截断
  return [...candidates.values()]
    .sort((a, b) => b.weight - a.weight || priorityOrder(a.entry.priority) - priorityOrder(b.entry.priority))
    .slice(0, limit);
}
```

**代码示例**：
```javascript
// 10字段加权搜索核心
const SEARCH_FIELDS = [
  { key: 'consumerQ', weight: 10, match: 'fuzzy' },
  { key: 'ownerQ', weight: 10, match: 'fuzzy' },
  { key: 'name', weight: 8, match: 'exact+fuzzy' },
  { key: 'alias', weight: 7, match: 'exact+fuzzy', isArray: true },
  { key: 'oneLineAnswer', weight: 5, match: 'fuzzy' },
  { key: 'corePoint', weight: 4, match: 'fuzzy', isArray: true },
  { key: 'def', weight: 2, match: 'fulltext' },
  { key: 'legalRef', weight: 2, match: 'exact', isArray: true },
  { key: 'subScene', weight: 1, match: 'exact' },
  { key: 'entryType', weight: 1, match: 'exact' }
];

const PRIORITY_BONUS = { P0: 2, P1: 1.5, P2: 1, P3: 0.5 };
const SEVERITY_BONUS = { hard: 1, medium: 0.5, soft: 0 };

function computeMatchScore(entry, query) {
  let totalScore = 0;
  let bestMatchField = null;
  let bestMatchDegree = 0;

  for (const field of SEARCH_FIELDS) {
    const values = field.isArray
      ? (Array.isArray(entry[field.key]) ? entry[field.key] : [entry[field.key]])
      : [entry[field.key]];

    for (const val of values) {
      if (!val) continue;
      const str = String(val).toLowerCase();
      const q = query.toLowerCase();
      let degree = 0;
      if (str === q) degree = 1.0;              // 精确匹配
      else if (str.startsWith(q)) degree = 0.8;  // 前缀匹配
      else if (str.includes(q)) degree = 0.5;    // 包含匹配
      // 模糊匹配可后续扩展（编辑距离/拼音）

      if (degree > 0) {
        totalScore += field.weight * degree;
        if (degree > bestMatchDegree) {
          bestMatchDegree = degree;
          bestMatchField = field.key;
        }
      }
    }
  }

  // 优先级+严重度加成
  totalScore += PRIORITY_BONUS[entry.priority] || 0;
  totalScore += SEVERITY_BONUS[entry.severity] || 0;

  return { score: totalScore, matchField: bestMatchField, matchDegree: bestMatchDegree };
}
```

```javascript
if (path.startsWith('/api/v1/')) {

  // 场景详情
  if (path.match(/^\/api\/v1\/scene\/[^/]+\/[^/]+\/detail$/)) return handleV1SceneDetail(request, env, ctx);
  // 词条详情
  if (path.match(/^\/api\/v1\/entry\/[^/]+$/) && request.method === 'GET') return handleV1EntryDetail(request, env, ctx);
  // 词条关联
  if (path.match(/^\/api\/v1\/entry\/[^/]+\/related$/)) return handleV1EntryRelated(request, env, ctx);
  // 全量词典
  if (path === '/api/v1/dictionary') return handleV1Dictionary(request, env, ctx);
  // 搜索
  if (path === '/api/v1/search') return handleV1Search(request, env, ctx);
  // 搜索联想
  if (path === '/api/v1/search/suggest') return handleV1SearchSuggest(request, env, ctx);
  // 小程序-每日速递
  if (path === '/api/v1/mini/daily') return handleV1MiniDaily(request, env, ctx);
  // 小程序-场景词条
  if (path === '/api/v1/mini/scene/entries') return handleV1MiniSceneEntries(request, env, ctx);
  // 小程序-词条详情
  if (path.match(/^\/api\/v1\/mini\/entry\/[^/]+$/)) return handleV1MiniEntryDetail(request, env, ctx);
  // 未匹配的v1路由
  return jsonResponse({ code: 1004, message: '未知的v1接口' }, 404);
}
```
